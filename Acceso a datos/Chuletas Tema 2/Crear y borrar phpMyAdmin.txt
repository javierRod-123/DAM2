CREATE DATABASE IF NOT EXISTS gimnasio
  CHARACTER SET utf8mb4
  COLLATE utf8mb4_unicode_ci;

CREATE USER IF NOT EXISTS 'gimnasio_user'@'localhost' IDENTIFIED BY 'gimnasio_pass';
CREATE USER IF NOT EXISTS 'gimnasio_user'@'%' IDENTIFIED BY 'gimnasio_pass';

GRANT ALL PRIVILEGES ON gimnasio.* TO 'gimnasio_user'@'localhost';
GRANT ALL PRIVILEGES ON gimnasio.* TO 'gimnasio_user'@'%';

FLUSH PRIVILEGES;
